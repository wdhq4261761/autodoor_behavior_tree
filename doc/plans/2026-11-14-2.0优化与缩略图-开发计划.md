```python
"""运行前静态预检

纯函数模块：不依赖 GUI，可在 GUI / CLI / Headless 中复用。
必须在节点注册（含插件）完成之后调用，否则插件节点会被误判为未知类型。
"""
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional

from bt_core.registry import NodeRegistry

# 需要校验资源存在的节点类型 -> (配置键, 说明)
RESOURCE_FIELDS = {
    "ImageConditionNode": ("template_path", "模板图片"),
    "SubtreeNode": ("subtree_path", "子树文件"),
    "ScriptNode": ("script_path", "脚本文件"),
    "TextInputNode": ("file_path", "输入文件"),
}


@dataclass(frozen=True)
class PreflightIssue:
    level: str          # "error" | "warning"
    code: str
    message: str
    node_id: Optional[str] = None


@dataclass
class PreflightReport:
    issues: List[PreflightIssue] = field(default_factory=list)

    @property
    def has_errors(self) -> bool:
        return any(i.level == "error" for i in self.issues)

    def add(self, level: str, code: str, message: str, node_id: Optional[str] = None) -> None:
        self.issues.append(PreflightIssue(level, code, message, node_id))

    def format_lines(self) -> List[str]:
        prefix = {"error": "[错误]", "warning": "[警告]"}
        return [
            f"{prefix.get(i.level, '[提示]')} {i.message}"
            + (f"（节点 {i.node_id}）" if i.node_id else "")
            for i in self.issues
        ]


def _normalize_connections(tree_data: Dict[str, Any]) -> List[Dict[str, str]]:
    conns = tree_data.get("connections") or []
    result = []
    for c in conns:
        if isinstance(c, dict):
            result.append({"parent_id": c.get("parent_id"), "child_id": c.get("child_id")})
    return result


def _resolve_path(path: str, project_root: Optional[str]) -> str:
    if not path:
        return ""
    if path.startswith("./") and project_root:
        return os.path.normpath(os.path.join(project_root, path[2:]))
    if os.path.isabs(path):
        return path
    if project_root:
        return os.path.normpath(os.path.join(project_root, path))
    return os.path.abspath(path)


def validate_tree_data(tree_data: Dict[str, Any], project_root: Optional[str] = None) -> PreflightReport:
    import os

    report = PreflightReport()
    nodes = tree_data.get("nodes") or {}
    if not nodes:
        report.add("error", "EMPTY_TREE", "行为树为空，至少需要一个开始节点")
        return report

    root_id = tree_data.get("root_node")
    if not root_id or root_id not in nodes:
        report.add("error", "ROOT_MISSING", "未找到有效的根节点（root_node）")

    conns = _normalize_connections(tree_data)
    parents: Dict[str, List[str]] = {}
    children: Dict[str, List[str]] = {}
    for c in conns:
        p, ch = c["parent_id"], c["child_id"]
        if p not in nodes or ch not in nodes:
            report.add("error", "BAD_CONNECTION", f"连线引用了不存在的节点：{p} -> {ch}")
            continue
        parents.setdefault(ch, []).append(p)
        children.setdefault(p, []).append(ch)

    for node_id, ps in parents.items():
        if len(ps) > 1:
            report.add("error", "MULTI_PARENT",
                       f"节点被多个父节点引用：{', '.join(ps)}", node_id)

    for node_id, node in nodes.items():
        node_type = node.get("type", "")
        if NodeRegistry.get(node_type) is None:
            report.add("error", "UNKNOWN_NODE_TYPE", f"未知节点类型：{node_type}", node_id)
            continue

        config = node.get("config") or {}
        resource = RESOURCE_FIELDS.get(node_type)
        if resource:
            key, label = resource
            value = config.get(key, "")
            if value and not os.path.exists(_resolve_path(str(value), project_root)):
                report.add("error", "MISSING_RESOURCE",
                           f"{label}不存在：{value}", node_id)
            elif not value:
                report.add("warning", "EMPTY_RESOURCE",
                           f"{label}未设置", node_id)

    # 环检测（三色 DFS）
    color: Dict[str, int] = {}

    def dfs(nid: str) -> bool:
        color[nid] = 1
        for child in children.get(nid, []):
            if color.get(child) == 1:
                return False
            if color.get(child, 0) == 0 and not dfs(child):
                return False
        color[nid] = 2
        return True

    for nid in nodes:
        if color.get(nid, 0) == 0 and not dfs(nid):
            report.add("error", "CYCLE", "行为树存在循环引用")
            break

    if root_id in nodes:
        reachable = set()

        def walk(nid: str) -> None:
            if nid in reachable:
                return
            reachable.add(nid)
            for child in children.get(nid, []):
                walk(child)

        walk(root_id)
        for nid in nodes:
            if nid not in reachable:
                report.add("warning", "UNREACHABLE", "节点不可达（从开始节点无法到达）", nid)

    return report
```

def mouse_scroll(self, amount: int, position: tuple = None, horizontal: bool = False) -> None:
    import pyautogui
    if position:

def execute_mouse_scroll(self, amount: int, position: tuple = None, horizontal: bool = False) -> bool:
    try:
        manager = self._get_input_manager()
        kwargs = {}
        if manager.get_mouse_method() == "bg" and self._bound_window:
            kwargs["hwnd"] = self._bound_window
        engine = manager.get_mouse_engine(**kwargs)
        if not engine:
            return False
        engine.mouse_scroll(amount, position, horizontal=horizontal)
        return True
    except Exception as e:
        from bt_utils.log_manager import LogManager
        LogManager.debug_print(f"[WARN] mouse_scroll failed: {e}")
        return False

@staticmethod
def parse_window_size(text: str):
    if not text:
        return None
    normalized = str(text).lower().replace("×", "x").replace("*", "x").replace(" ", "")
    parts = normalized.split("x")
    if len(parts) != 2:
        return None
    try:
        w, h = int(parts[0]), int(parts[1])
    except ValueError:
        return None
    return (w, h) if w > 0 and h > 0 else None

@staticmethod
def format_window_size(size) -> str:
    if not size:
        return ""
    return f"{int(size[0])}x{int(size[1])}"

@staticmethod
def get_window_size(hwnd: int):
    import win32gui
    try:
        left, top, right, bottom = win32gui.GetWindowRect(hwnd)
        return (right - left, bottom - top)
    except Exception:
        return None

@staticmethod
def set_window_size(hwnd: int, size) -> bool:
    import win32gui
    try:
        left, top, _, _ = win32gui.GetWindowRect(hwnd)
        win32gui.SetWindowPos(hwnd, None, left, top, int(size[0]), int(size[1]),
                              0x0002 | 0x0004 | 0x0010)  # SWP_NOMOVE|SWP_NOZORDER|SWP_NOACTIVATE
        return True
    except Exception:
        return False

@staticmethod
def is_maximized(hwnd: int) -> bool:
    import win32gui
    try:
        return bool(win32gui.IsZoomed(hwnd))
    except Exception:
        return False
```

"""统一 JSON 读写：读取兼容 UTF-8 BOM（utf-8-sig），写入固定 UTF-8。"""
import json
import os
from typing import Any, Optional


def load_json(path, default: Optional[Any] = None) -> Any:
    try:
        with open(path, "r", encoding="utf-8-sig") as f:
            return json.load(f)
    except FileNotFoundError:
        if default is not None:
            return default
        raise
    except json.JSONDecodeError as e:
        raise ValueError(f"JSON 格式错误: {path}: {e}") from e


def save_json(path, data: Any, indent: int = 2) -> None:
    parent = os.path.dirname(os.path.abspath(path))
    if parent:
        os.makedirs(parent, exist_ok=True)
    with open(path, "w", encoding="utf-8", newline="\n") as f:
        json.dump(data, f, ensure_ascii=False, indent=indent)
```

替换规则：
- `with open(x, 'r', encoding='utf-8') as f: data = json.load(f)` → `data = load_json(x)`
- 需要保留“文件不存在返回默认值”的原逻辑时传 `default=...`
- 写入若原本是 `json.dump(..., ensure_ascii=False, indent=2)`，可保持原样，不强制替换（本任务只保证读兼容 BOM）

```python
"""模板缩略图：纯函数，便于单测，不依赖 Tk。"""
import os

from PIL import Image

DEFAULT_MAX_SIZE = (232, 140)


def resolve_resource_path(path: str, project_root: str = None) -> str:
    if not path:
        return ""
    if path.startswith("./") and project_root:
        return os.path.normpath(os.path.join(project_root, path[2:]))
    if os.path.isabs(path):
        return path
    if project_root:
        return os.path.normpath(os.path.join(project_root, path))
    return os.path.abspath(path)


def load_thumbnail(path: str, max_size=DEFAULT_MAX_SIZE):
    if not path or not os.path.isfile(path):
        return None, None
    try:
        with Image.open(path) as img:
            img.load()
            img = img.convert("RGBA")
            width, height = img.size
            max_w, max_h = max_size
            ratio = min(max_w / width, max_h / height, 1.0)
            size = (max(1, int(width * ratio)), max(1, int(height * ratio)))
            return img.resize(size, Image.LANCZOS), size
    except Exception:
        return None, None
```

|---|---|---|
| 预检误报（插件节点未注册） | 无法运行 | 预检必须在插件加载后执行；未知类型只报 error 不抛异常；GUI 允许确认继续 |
| 窗口还原改变用户窗口习惯 | 体验 | 仅当已有记录且差异 >2px 才还原；提供开关；最大化不主动还原 |

| 拖拽补点导致双重点击 | 误操作 | 仅在鼠标松开前补移动，不加点击 |
| 缩略图大图导致卡顿 | UI 卡顿 | 只读取并缩放到 232x140；异步 step 中创建；异常降级为文字 |
| 全量回归失败 | 阻塞发布 | 每个 Task 独立提交，可单独 revert |
